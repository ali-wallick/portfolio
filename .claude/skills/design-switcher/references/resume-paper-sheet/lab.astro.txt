/**
 * SCAFFOLDING — not shipped. Deleted before this branch merges.
 *
 * `/design/resume-print`: the résumé rendered AS THE PRINTED PAGE, on screen,
 * behind a switcher — issue #235. A PDF is a still frame, so the comparison
 * that carries information is flipping treatments on the same sheet, not
 * opening N PDFs in N tabs.
 *
 * The sheet reproduces `resume.css`'s `@media print` cascade on screen:
 * letter geometry (816px wide, 0.5in/0.6in margins, a 701px content column),
 * the print block's token pins and rules transcribed under `.lab-sheet`, and
 * a dashed line wherever a page boundary falls. The panel reports the content
 * height against the 960px-per-page budget live, because the one-pager sits
 * at 928px today (32px of slack — `npm run resume:headroom`) and a candidate
 * that overflows is dead however it looks.
 *
 * `noindex`, absent from `src/pages/sitemap.xml.ts`, and a dynamic route
 * returning no paths unless `showDrafts` — so a production build emits no
 * `dist/design/` at all (see the design-switcher skill: uncrawled is not
 * unserved).
 *
 * Nothing on this route is a `byteHashedFiles` input. `resume.css` is not
 * imported here on purpose: its screen half would style the sheet as the
 * panel, and its print half is what this stylesheet transcribes.
 */
export const getStaticPaths = (() =>
  showDrafts ? [{ params: { lab: 'resume-print' } }] : []) satisfies GetStaticPaths;

/* Every axis lists the incumbent first, as the labelled baseline. Labels
